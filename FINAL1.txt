    function decodeJwtResponse(token) {
      try {
        const base64Url = token.split('.')[1];
        const base64 = base64Url.replace(/-/g, '+').replace(/_/g, '/');
        const jsonPayload = decodeURIComponent(window.atob(base64).split('').map(function(c) {
            return '%' + ('00' + c.charCodeAt(0).toString(16)).slice(-2);
        }).join(''));
        return JSON.parse(jsonPayload);
      } catch (e) {
        console.error("JWT Decode error:", e);
        return null;
      }
    }

    function handleCredentialResponse(response) {
      const responsePayload = decodeJwtResponse(response.credential);
      if (responsePayload) {
        localStorage.setItem('yash_user', JSON.stringify({
          name: responsePayload.name,
          email: responsePayload.email,
          picture: responsePayload.picture
        }));
        updateAuthUI(responsePayload);
      }
    }

    function updateAuthUI(user) {
      const signInContainer = document.getElementById('g_id_signin_container');
      const profileContainer = document.getElementById('user-profile');
      const avatar = document.getElementById('user-avatar');
      const name = document.getElementById('user-name');

      if (user) {
        signInContainer.classList.add('hidden');
        profileContainer.classList.remove('hidden');
        profileContainer.classList.add('flex');
        avatar.src = user.picture || 'https://via.placeholder.com/32';
        name.innerText = user.name || user.email || 'User';
      } else {
        profileContainer.classList.add('hidden');
        profileContainer.classList.remove('flex');
        signInContainer.classList.remove('hidden');
      }
    }

    function handleSignOut() {
      localStorage.removeItem('yash_user');
      if (window.google && google.accounts.id) {
        google.accounts.id.disableAutoSelect();
      }
      updateAuthUI(null);
    }

    window.addEventListener('DOMContentLoaded', () => {
      const saved = localStorage.getItem('yash_user');
      if (saved) {
        try {
          updateAuthUI(JSON.parse(saved));
        } catch (e) {
          localStorage.removeItem('yash_user');
        }
      }
    });

      let i = 0;

    function renderRadarChart(intels) {
      if (radarChartInstance) radarChartInstance.destroy();
      const ctx = document.getElementById('radarChart').getContext('2d');
      const sorted = [...intels].sort((a,b) => a.idx - b.idx);
      const grad = ctx.createLinearGradient(0, 0, 0, 400);
      grad.addColorStop(0, 'rgba(211, 175, 55, 0.7)');
      grad.addColorStop(1, 'rgba(238, 225, 173, 0.15)');

      radarChartInstance = new Chart(ctx, {
        type: 'radar',
        data: {
          labels: sorted.map(i => i.name),
          datasets: [{ data: sorted.map(i => i.score), backgroundColor: grad, borderColor: '#d3af37', pointBackgroundColor: '#eee1ad', pointBorderColor: '#1a1a1a', borderWidth: 2, tension: 0.3 }]
        },
        options: {
          responsive: true, maintainAspectRatio: false,
          scales: { r: { angleLines: { color: 'rgba(211, 175, 55, 0.2)' }, grid: { color: 'rgba(211, 175, 55, 0.15)', circular: true }, pointLabels: { color: '#eee1ad', font: { size: 10, family: 'Inter', weight: '600' } }, ticks: { display: false, min: 0, max: Math.max(...sorted.map(i=>i.score)) + 1 } } },
          plugins: { legend: { display: false } }
        }
      });
    }
